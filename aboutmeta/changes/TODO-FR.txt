from ruamel.yaml import YAML

def get_global_comment(mapping):
    """Récupère le commentaire global situé tout en haut du fichier YAML."""
    if not hasattr(mapping, 'ca') or not mapping.ca.comment:
        return None

    comment_list = mapping.ca.comment[1]
    if comment_list:
        lines = [token.value.strip() for token in comment_list if token is not None]
        return "\n".join(lines)
    return None

def get_comment_for_key(mapping, key):
    """Récupère le commentaire associé à une clé (au-dessus ou en fin de ligne)."""
    if not hasattr(mapping, 'ca') or key not in mapping.ca.items:
        return None

    item = mapping.ca.items[key]

    # 1. Priorité au commentaire en fin de ligne (Inline comment: ex: port: 8080 # text)
    if item[2] is not None:
        return item[2].value.strip()

    # 2. Sinon, commentaire sur la/les ligne(s) précédente(s) (Block comment)
    if item[1] is not None:
        lines = [token.value.strip() for token in item[1] if token is not None]
        return "\n".join(lines)

    return None

> ***NOTE.*** *Using the `Python` API or the CLI, you can request the addition of a `License.txt` file in the folder containing the `about.yaml` file.*




> ***NOTE.*** *The CLI allows to request that the full text of the license be put into a file `LICENSE.txt `relatively to the folder containing the `about.yaml` file. You can also do that using codes like the follwing one.*

~~~python
from aboutmeta import AMData, Path

meta = AMData(Path("/full/path/to/about.yaml"))
meta.build()

meta.add_license(
    license_path = "project.licenses.code",
    erase        = True
)

meta.add_license(
    license_path = "project.licenses.manual",
    dir_relpath  = "readme",
    erase        = True
)
~~~
